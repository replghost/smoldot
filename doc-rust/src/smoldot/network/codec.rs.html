48/// Name of a protocol that is part of the Substrate/Polkadot networking.
49#[derive(Copy, Clone, PartialEq, Eq, Hash)]
50pub enum ProtocolName<'a> {
51    Identify,
52    Ping,
53    BlockAnnounces {
54        genesis_hash: [u8; 32],
55        fork_id: Option<&'a str>,
56    },
57    Transactions {
58        genesis_hash: [u8; 32],
59        fork_id: Option<&'a str>,
60    },
61    Grandpa {
62        genesis_hash: [u8; 32],
63        fork_id: Option<&'a str>,
64    },
65    Sync {
66        genesis_hash: [u8; 32],
67        fork_id: Option<&'a str>,
68    },
69    Light {
70        genesis_hash: [u8; 32],
71        fork_id: Option<&'a str>,
72    },
73    Kad {
74        genesis_hash: [u8; 32],
75        fork_id: Option<&'a str>,
76    },
77    SyncWarp {
78        genesis_hash: [u8; 32],
79        fork_id: Option<&'a str>,
80    },
81    State {
82        genesis_hash: [u8; 32],
83        fork_id: Option<&'a str>,
84    },
85    Statement {
86        genesis_hash: [u8; 32],
87        fork_id: Option<&'a str>,
88    },
89}

91impl<'a> fmt::Debug for ProtocolName<'a> {
92    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
93        fmt::Display::fmt(self, f)
94    }
95}

97impl<'a> fmt::Display for ProtocolName<'a> {
98    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
99        for chunk in encode_protocol_name(*self) {
100            f.write_str(chunk.as_ref())?;
101        }
102        Ok(())
103    }
104}

106/// Turns a [`ProtocolName`] into its string version. Returns a list of objects that, when
107/// concatenated together, forms the string version of the [`ProtocolName`].
108pub fn encode_protocol_name(protocol: ProtocolName) -> impl Iterator<Item = impl AsRef<str>> {
109    let (genesis_hash, fork_id, base_protocol_name) = match protocol {
110        ProtocolName::Identify => return either::Left(iter::once(Cow::Borrowed("/ipfs/id/1.0.0"))),
111        ProtocolName::Ping => return either::Left(iter::once(Cow::Borrowed("/ipfs/ping/1.0.0"))),
112        ProtocolName::BlockAnnounces {
113            genesis_hash,
114            fork_id,
115        } => (genesis_hash, fork_id, "block-announces/1"),
116        ProtocolName::Transactions {
117            genesis_hash,
118            fork_id,
119        } => (genesis_hash, fork_id, "transactions/1"),
120        ProtocolName::Grandpa {
121            genesis_hash,
122            fork_id,
123        } => (genesis_hash, fork_id, "grandpa/1"),
124        ProtocolName::Sync {
125            genesis_hash,
126            fork_id,
127        } => (genesis_hash, fork_id, "sync/2"),
128        ProtocolName::Light {
129            genesis_hash,
130            fork_id,
131        } => (genesis_hash, fork_id, "light/2"),
132        ProtocolName::Kad {
133            genesis_hash,
134            fork_id,
135        } => (genesis_hash, fork_id, "kad"),
136        ProtocolName::SyncWarp {
137            genesis_hash,
138            fork_id,
139        } => (genesis_hash, fork_id, "sync/warp"),
140        ProtocolName::State {
141            genesis_hash,
142            fork_id,
143        } => (genesis_hash, fork_id, "state/2"),
144        ProtocolName::Statement {
145            genesis_hash,
146            fork_id,
147        } => (genesis_hash, fork_id, "statement/1"),
148    };
149
150    let genesis_hash = hex::encode(genesis_hash);
151
152    if let Some(fork_id) = fork_id {
153        either::Right(either::Right(
154            [
155                Cow::Borrowed("/"),
156                Cow::Owned(genesis_hash),
157                Cow::Borrowed("/"),
158                Cow::Borrowed(fork_id),
159                Cow::Borrowed("/"),
160                Cow::Borrowed(base_protocol_name),
161            ]
162            .into_iter(),
163        ))
164    } else {
165        either::Right(either::Left(
166            [
167                Cow::Borrowed("/"),
168                Cow::Owned(genesis_hash),
169                Cow::Borrowed("/"),
170                Cow::Borrowed(base_protocol_name),
171            ]
172            .into_iter(),
173        ))
174    }
175}

177/// Turns a [`ProtocolName`] into a string.
178pub fn encode_protocol_name_string(protocol: ProtocolName) -> String {
179    encode_protocol_name(protocol).fold(String::with_capacity(128), |mut a, b| {
180        a.push_str(b.as_ref());
181        a
182    })
183}

185/// Decodes a protocol name into its components.
186///
187/// Returns an error if the protocol name isn't recognized.
188pub fn decode_protocol_name(name: &'_ str) -> Result<ProtocolName<'_>, ()> {
189    nom::Parser::parse(
190        &mut nom::combinator::all_consuming(nom::branch::alt((
191            nom::combinator::map(nom::bytes::complete::tag("/ipfs/id/1.0.0"), |_| {
192                ProtocolName::Identify
193            }),
194            nom::combinator::map(nom::bytes::complete::tag("/ipfs/ping/1.0.0"), |_| {
195                ProtocolName::Ping
196            }),
197            nom::combinator::map(
198                (
199                    nom::bytes::complete::tag("/"),
200                    genesis_hash,
201                    nom::bytes::complete::tag("/"),
202                    protocol_ty,
203                ),
204                |(_, genesis_hash, _, protocol_ty)| {
205                    protocol_ty_to_real_protocol(protocol_ty, genesis_hash, None)
206                },
207            ),
208            nom::combinator::map(
209                (
210                    nom::bytes::complete::tag("/"),
211                    genesis_hash,
212                    nom::bytes::complete::tag("/"),
213                    nom::bytes::complete::take_until("/"),
214                    nom::bytes::complete::tag("/"),
215                    protocol_ty,
216                ),
217                |(_, genesis_hash, _, fork_id, _, protocol_ty)| {
218                    protocol_ty_to_real_protocol(protocol_ty, genesis_hash, Some(fork_id))
219                },
220            ),
221        ))),
222        name,
223    )
224    .map(|(_, parse_result)| parse_result)
225    .map_err(|_| ())
226}
227
228fn genesis_hash(name: &str) -> nom::IResult<&str, [u8; 32]> {
229    nom::Parser::parse(
230        &mut nom::combinator::map_opt(nom::bytes::complete::take(64u32), |hash| {
231            hex::decode(hash)
232                .ok()
233                .map(|hash| <[u8; 32]>::try_from(hash).unwrap_or_else(|_| unreachable!()))
234        }),
235        name,
236    )
237}

239enum ProtocolTy {
240    BlockAnnounces,
241    Transactions,
242    Grandpa,
243    Sync,
244    Light,
245    Kad,
246    SyncWarp,
247    State,
248    Statement,
249}
250
251fn protocol_ty(name: &str) -> nom::IResult<&str, ProtocolTy> {
252    nom::Parser::parse(
253        &mut nom::branch::alt((
254            nom::combinator::map(nom::bytes::complete::tag("block-announces/1"), |_| {
255                ProtocolTy::BlockAnnounces
256            }),
257            nom::combinator::map(nom::bytes::complete::tag("transactions/1"), |_| {
258                ProtocolTy::Transactions
259            }),
260            nom::combinator::map(nom::bytes::complete::tag("grandpa/1"), |_| {
261                ProtocolTy::Grandpa
262            }),
263            nom::combinator::map(nom::bytes::complete::tag("sync/2"), |_| ProtocolTy::Sync),
264            nom::combinator::map(nom::bytes::complete::tag("light/2"), |_| ProtocolTy::Light),
265            nom::combinator::map(nom::bytes::complete::tag("kad"), |_| ProtocolTy::Kad),
266            nom::combinator::map(nom::bytes::complete::tag("sync/warp"), |_| {
267                ProtocolTy::SyncWarp
268            }),
269            nom::combinator::map(nom::bytes::complete::tag("state/2"), |_| ProtocolTy::State),
270            nom::combinator::map(nom::bytes::complete::tag("statement/1"), |_| {
271                ProtocolTy::Statement
272            }),
273        )),
274        name,
275    )
276}

278fn protocol_ty_to_real_protocol(
279    ty: ProtocolTy,
280    genesis_hash: [u8; 32],
281    fork_id: Option<&'_ str>,
282) -> ProtocolName<'_> {
283    match ty {
284        ProtocolTy::BlockAnnounces => ProtocolName::BlockAnnounces {
285            genesis_hash,
286            fork_id,
287        },
288        ProtocolTy::Transactions => ProtocolName::Transactions {
289            genesis_hash,
290            fork_id,
291        },
292        ProtocolTy::Grandpa => ProtocolName::Grandpa {
293            genesis_hash,
294            fork_id,
295        },
296        ProtocolTy::Sync => ProtocolName::Sync {
297            genesis_hash,
298            fork_id,
299        },
300        ProtocolTy::Light => ProtocolName::Light {
301            genesis_hash,
302            fork_id,
303        },
304        ProtocolTy::Kad => ProtocolName::Kad {
305            genesis_hash,
306            fork_id,
307        },
308        ProtocolTy::SyncWarp => ProtocolName::SyncWarp {
309            genesis_hash,
310            fork_id,
311        },
312        ProtocolTy::State => ProtocolName::State {
313            genesis_hash,
314            fork_id,
315        },
316        ProtocolTy::Statement => ProtocolName::Statement {
317            genesis_hash,
318            fork_id,
319        },
320    }
321}

323#[cfg(test)]
324mod tests {
325    use super::*;
326
327    fn encode_to_string(protocol: ProtocolName) -> String {
328        encode_protocol_name(protocol)
329            .map(|s| s.as_ref().to_owned())
330            .collect::<Vec<_>>()
331            .join("")
332    }
333
334    #[test]
335    fn statement_protocol_name() {
336        let decoded = decode_protocol_name(
337            "/fccae9f32db7eedb8310800252752bf649e2e3661a9260f784b3579ce14933af/statement/1",
338        )
339        .unwrap();
340
341        assert!(matches!(
342            decoded,
343            ProtocolName::Statement { genesis_hash, fork_id: None }
344            if hex::encode(genesis_hash) == "fccae9f32db7eedb8310800252752bf649e2e3661a9260f784b3579ce14933af"
345        ));
346    }
347
348    #[test]
349    fn statement_protocol_name_roundtrip_with_fork() {
350        let genesis_hash = [0xab; 32];
351        let protocol = ProtocolName::Statement {
352            genesis_hash,
353            fork_id: Some("polkadot"),
354        };
355
356        let encoded = encode_to_string(protocol);
357        let decoded = decode_protocol_name(&encoded).unwrap();
358
359        assert!(matches!(
360            decoded,
361            ProtocolName::Statement { genesis_hash: gh, fork_id: Some("polkadot") }
362            if gh == genesis_hash
363        ));
364    }
